function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var action = data.action;
    var userId = data.userId;
    var email = data.email;
    var datos = data.datos;
    var resultado;

    switch (action) {
      // -- FUNCIONES NUEVAS SAAS --
      case 'obtenerUsuariosSaaS': resultado = obtenerUsuariosSaaS(); break;
      case 'actualizarUsuarioSaaS': resultado = actualizarUsuarioSaaS(datos); break;
      case 'obtenerDatosFinanzasSaaS': resultado = obtenerDatosFinanzasSaaS(); break;
      case 'registrarPagoSaaS': resultado = registrarPagoSaaS(datos); break;
      case 'obtenerMiSuscripcion': resultado = obtenerMiSuscripcion(userId); break;
      
      // -- FUNCIONES ORIGINALES MANTENIDAS INTACTAS --
      case 'verificarAccesoUsuario': resultado = verificarAccesoUsuario(userId, email); break;
      case 'obtenerDatosPerfil': resultado = obtenerDatosPerfil(); break;
      case 'actualizarDatosPerfilCompleto': resultado = actualizarDatosPerfilCompleto(datos); break;
      case 'obtenerListaClientes': resultado = obtenerListaClientes(); break;
      case 'guardarNuevoCliente': resultado = guardarNuevoCliente(datos); break;
      case 'eliminarClienteServidor': resultado = eliminarClienteServidor(datos.dni); break;
      case 'obtenerDatosDashboard': resultado = obtenerDatosDashboard(); break;
      case 'obtenerDatosProyectos': resultado = obtenerDatosProyectos(); break;
      case 'guardarCambioProyecto': resultado = guardarCambioProyecto(datos.index, datos.tipo, datos.valor); break;
      case 'obtenerScripts': resultado = obtenerScripts(); break;
      case 'actualizarScripts': resultado = actualizarScripts(datos); break;
      case 'ejecutarEnvioDesdeApp': resultado = ejecutarEnvioDesdeApp(); break;
      case 'obtenerDatosReporteAnual': resultado = obtenerDatosReporteAnual(datos.anio); break;
      case 'obtenerDatosReporteClientes': resultado = obtenerDatosReporteClientes(datos.anio, datos.mesTexto); break;
      default: resultado = { error: "Acción no encontrada: " + action };
    }

    return ContentService.createTextOutput(JSON.stringify(resultado))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * ==========================================
 * LÓGICA DE AUTENTICACIÓN Y COPIA DE PLANTILLAS (MODIFICADA PARA ROLES)
 * ==========================================
 */
function verificarAccesoUsuario(uid, email) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dbSheet = ss.getSheetByName("DB_USUARIOS");
  if (!dbSheet) return { error: "Falta crear la pestaña DB_USUARIOS en tu Google Sheet maestro." };

  var data = dbSheet.getDataRange().getValues();
  var targetRow = -1;
  var sheetId = "";
  var estado = "";
  var rol = "";

  // Busca si el usuario ya existe (NUEVA ESTRUCTURA DE COLUMNAS)
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] == uid) {
      targetRow = i + 1;
      email = data[i][1];
      rol = data[i][3] || "Cliente";
      estado = data[i][4] || "Pendiente";
      sheetId = data[i][7]; // Col H: ID del Sheet
      break;
    }
  }

  // Si no existe, lo registra
  if (targetRow === -1) {
    targetRow = dbSheet.getLastRow() + 1;
    rol = (uid === ADMIN_UID) ? "Administrador" : "Cliente";
    estado = (uid === ADMIN_UID) ? "Autorizado" : "Pendiente";
    
    dbSheet.getRange(targetRow, 1).setValue(uid);
    dbSheet.getRange(targetRow, 2).setValue(email);
    dbSheet.getRange(targetRow, 4).setValue(rol); // Columna Rol
    dbSheet.getRange(targetRow, 5).setValue(estado); // Columna Estado
    dbSheet.getRange(targetRow, 6).setValue(Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd"));
  }

  // Si no tiene un Sheet asignado Y está autorizado, ejecuta la lógica de clonación
  if ((!sheetId || sheetId === "") && estado === "Autorizado") {
    if (uid === ADMIN_UID) {
      // El admin edita la plantilla maestra directamente
      sheetId = ID_PLANTILLA_MAESTRA;
      dbSheet.getRange(targetRow, 7).setValue("https://docs.google.com/spreadsheets/d/" + sheetId);
      dbSheet.getRange(targetRow, 8).setValue(sheetId);
    } else {
      // A los usuarios se les crea una copia
      var template = DriveApp.getFileById(ID_PLANTILLA_MAESTRA);
      var targetFolder = DriveApp.getFolderById(ID_CARPETA_DESTINO);
      var copy = template.makeCopy("Dashboard Freelancer - " + (email || uid), targetFolder);
      sheetId = copy.getId();
      dbSheet.getRange(targetRow, 7).setValue("https://docs.google.com/spreadsheets/d/" + sheetId);
      dbSheet.getRange(targetRow, 8).setValue(sheetId);
    }
  }

  return { sheetId: sheetId, estado: estado, rol: rol, isAdmin: (rol === "Administrador" || uid === ADMIN_UID) };
}

/**
 * ==========================================
 * NUEVAS FUNCIONES DE ADMINISTRACIÓN SAAS
 * ==========================================
 */
function obtenerUsuariosSaaS() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const dbSheet = ss.getSheetByName("DB_USUARIOS");
  if (!dbSheet) return [];
  
  const data = dbSheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  data.shift(); // Quitar cabeceras
  
  return data.map((fila, index) => {
    return {
      fila: index + 2, uid: fila[0], correo: fila[1], telefono: fila[2], rol: fila[3],
      estatus: fila[4], fecha: (fila[5] instanceof Date) ? Utilities.formatDate(fila[5], Session.getScriptTimeZone(), "yyyy-MM-dd") : fila[5]
    };
  }).filter(u => u.uid !== "");
}

function actualizarUsuarioSaaS(datos) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const dbSheet = ss.getSheetByName("DB_USUARIOS");
  dbSheet.getRange(datos.fila, 4).setValue(datos.rol);
  dbSheet.getRange(datos.fila, 5).setValue(datos.estatus);
  return "✅ Usuario actualizado";
}

function obtenerDatosFinanzasSaaS() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const dbSheet = ss.getSheetByName("DB_USUARIOS");
  const estadoSheet = ss.getSheetByName("ESTADO_SUSCRIPCIONES");

  let usuarios = [];
  let estadosMap = {};

  if (estadoSheet) {
    const dataE = estadoSheet.getDataRange().getValues();
    for (let i = 1; i < dataE.length; i++) {
      if(dataE[i][0]) estadosMap[dataE[i][0]] = { plan: dataE[i][1], proxPago: dataE[i][2], estado: dataE[i][3] };
    }
  }

  if (dbSheet) {
    const dataU = dbSheet.getDataRange().getValues();
    for (let i = 1; i < dataU.length; i++) {
      if(dataU[i][0] && dataU[i][4] === "Autorizado") { // Solo mapear autorizados
        usuarios.push({
          uid: dataU[i][0],
          correo: dataU[i][1],
          suscripcion: estadosMap[dataU[i][0]] || { plan: 'Sin plan', proxPago: '---', estado: 'Pendiente' }
        });
      }
    }
  }
  return usuarios;
}

function registrarPagoSaaS(datos) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let hPagos = ss.getSheetByName("HISTORIAL_PAGOS");
  let hEstado = ss.getSheetByName("ESTADO_SUSCRIPCIONES");

  const idPago = "PAGO-" + new Date().getTime();
  hPagos.appendRow([idPago, datos.uid, datos.fecha, datos.periodo, datos.plan, datos.metodo, datos.recibo]);

  // Cálculo del próximo mes o año
  let partes = datos.fecha.split("-");
  let d = new Date(partes[0], partes[1] - 1, partes[2]);
  if(datos.plan === 'Mensual') d.setMonth(d.getMonth() + 1);
  if(datos.plan === 'Anual') d.setFullYear(d.getFullYear() + 1);
  let nextDateStr = Utilities.formatDate(d, Session.getScriptTimeZone(), "yyyy-MM-dd");

  // Actualizar o crear en ESTADO_SUSCRIPCIONES
  let dataE = hEstado.getDataRange().getValues();
  let found = false;
  for(let i=1; i<dataE.length; i++){
    if(dataE[i][0] == datos.uid){
       hEstado.getRange(i+1, 2, 1, 3).setValues([[datos.plan, nextDateStr, "Al día"]]);
       found = true;
       break;
    }
  }
  if(!found){
    hEstado.appendRow([datos.uid, datos.plan, nextDateStr, "Al día"]);
  }
  return "✅ Pago registrado y mes sincronizado.";
}

function obtenerMiSuscripcion(uid) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let hPagos = ss.getSheetByName("HISTORIAL_PAGOS");
  let hEstado = ss.getSheetByName("ESTADO_SUSCRIPCIONES");

  let miEstado = { plan: 'Sin plan activo', proxPago: '---', estado: 'PENDIENTE' };
  let miHistorial = [];

  if (hEstado) {
    let dE = hEstado.getDataRange().getValues();
    for(let i=1; i<dE.length; i++){
      if(dE[i][0] == uid){
        miEstado = { plan: dE[i][1], proxPago: (dE[i][2] instanceof Date)? Utilities.formatDate(dE[i][2], Session.getScriptTimeZone(), "yyyy-MM-dd") : dE[i][2], estado: dE[i][3] };
        break;
      }
    }
  }

  if (hPagos) {
    let dP = hPagos.getDataRange().getValues();
    for(let i=1; i<dP.length; i++){
      if(dP[i][1] == uid){
        miHistorial.push({
          fecha: (dP[i][2] instanceof Date) ? Utilities.formatDate(dP[i][2], Session.getScriptTimeZone(), "yyyy-MM-dd") : dP[i][2],
          periodo: dP[i][3], plan: dP[i][4], metodo: dP[i][5], recibo: dP[i][6]
        });
      }
    }
  }
  return { estado: miEstado, historial: miHistorial };
}

function obtenerDatosPerfil() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hoja = ss.getSheetByName("MI PERFIL");
    if (!hoja) return { error: "No existe la hoja MI PERFIL" };
    
    const rawLogoUrl = hoja.getRange("C11").getValue();
    let finalLogoUrl = rawLogoUrl;

    if (rawLogoUrl && (rawLogoUrl.includes("drive.google.com") || rawLogoUrl.includes("google.com/open"))) {
      try {
        let id = "";
        if (rawLogoUrl.includes("id=")) {
          id = rawLogoUrl.split("id=")[1].split("&")[0];
        } else if (rawLogoUrl.includes("/d/")) {
          id = rawLogoUrl.split("/d/")[1].split("/")[0];

          id = rawLogoUrl.split("/file/d/")[1].split("/")[0];
        }
        if (id) finalLogoUrl = "https://drive.google.com/thumbnail?sz=w500&id=" + id;

    }
    
    return {
      nombre: hoja.getRange("C5").getValue(),
      marca: hoja.getRange("C7").getValue(),
      especialidad: hoja.getRange("C9").getValue(),
      logoUrl: finalLogoUrl, 
      moneda: hoja.getRange("C17").getValue(),
      meta: hoja.getRange("C19").getValue(),
      rs1_n: hoja.getRange("B23").getValue(), rs1_l: hoja.getRange("C23").getValue(),
      rs2_n: hoja.getRange("B25").getValue(), rs2_l: hoja.getRange("C25").getValue(),

    const ss = SpreadsheetApp.getActiveSpreadsheet();

    if (!hoja) return [];

}

/**
 * 4. DASHBOARD Y MÉTRICAS (ORIGINAL INTACTO)
 */
function actualizarCalculosDashboard() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const hojaDash = ss.getSheetByName("DASHBOARD");
  const hojaP = ss.getSheetByName("PROYECTOS Y FINANZAS");
  const hojaPerfil = ss.getSheetByName("MI PERFIL");
  const hojaMotor = ss.getSheetByName("MOTOR GRAFICOS");
  
  if (!hojaDash || !hojaP || !hojaPerfil || !hojaMotor) return;

  const hoy = new Date();
  const mesActual = hoy.getMonth();
  const anioActual = hoy.getFullYear();

  const nombrePerfil = hojaPerfil.getRange("C5").getValue().toString().toUpperCase();
  hojaDash.getRange("A1").setValue("DASHBOARD DE " + nombrePerfil);
  
  const nombreMes = hoy.toLocaleDateString('es-ES', { month: 'long' }).toUpperCase();
  hojaDash.getRange("A7").setValue(nombreMes + "-" + anioActual);

  const uF = hojaP.getLastRow();
  let totalCobradoMes = 0;
  let facturasPendMes = 0;
  let proyectosActivosMes = 0;
  let pendientesLista = []; 

  if (uF >= 3) {
    const datosP = hojaP.getRange(3, 1, uF - 2, 7).getValues(); 
    datosP.forEach(fila => {
      const nombreProyecto = fila[0];
      const cliente = fila[1];
      const monto = parseFloat(fila[2]) || 0;
      const cobrado = (fila[3] === true || fila[3].toString().toUpperCase() === "TRUE"); 
      const fecha = fila[6];

      if (fecha instanceof Date && !isNaN(fecha)) {
        if (fecha.getMonth() === mesActual && fecha.getFullYear() === anioActual) {
          proyectosActivosMes++;
          if (cobrado === true) totalCobradoMes += monto;
          else facturasPendMes += monto;
        }
      }

      if (nombreProyecto !== "" && cobrado === false && pendientesLista.length < 5) {
        pendientesLista.push([nombreProyecto, cliente, monto]);
      }
    });
  }

  hojaDash.getRange("B4").setValue(totalCobradoMes);
  hojaDash.getRange("E4").setValue(facturasPendMes);
  hojaDash.getRange("H4").setValue(proyectosActivosMes);

  hojaDash.getRange("B30:B34").clearContent();
  hojaDash.getRange("E30:E34").clearContent();
  hojaDash.getRange("H30:H34").clearContent();

  if (pendientesLista.length > 0) {
    hojaDash.getRange(30, 2, pendientesLista.length, 1).setValues(pendientesLista.map(p => [p[0]]));
    hojaDash.getRange(30, 5, pendientesLista.length, 1).setValues(pendientesLista.map(p => [p[1]]));
    hojaDash.getRange(30, 8, pendientesLista.length, 1).setValues(pendientesLista.map(p => [p[2]]));
  } else {
    hojaDash.getRange("B30").setValue("No hay pendientes");
  }

  const metaConfigurada = parseFloat(hojaPerfil.getRange("C19").getValue()) || 0;
  hojaDash.getRange("I24").setValue(metaConfigurada);
  hojaDash.getRange("H26").setValue(totalCobradoMes);

  if (totalCobradoMes > metaConfigurada) {
    hojaDash.getRange("I26").setValue("¡META SUPERADA!");
  } else {
    hojaDash.getRange("I26").setValue(metaConfigurada - totalCobradoMes);
  }

  const gananciaExtra = Math.max(0, totalCobradoMes - metaConfigurada);
  if (gananciaExtra > 0) {
    hojaDash.getRange("H27").setValue("¡FELICIDADES! HAS GANADO $" + gananciaExtra.toFixed(2) + " EXTRAS ESTE MES 🚀");
  } else {
    hojaDash.getRange("H27").setValue("");
  }

  hojaMotor.getRange("E3").setValue(Math.min(totalCobradoMes, metaConfigurada));
  hojaMotor.getRange("E4").setValue(Math.max(0, metaConfigurada - totalCobradoMes));

  SpreadsheetApp.flush();
}

function obtenerDatosDashboard() {
  try {
    actualizarCalculosDashboard();
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hojaDash = ss.getSheetByName("DASHBOARD");
    const hojaMotor = ss.getSheetByName("MOTOR GRAFICOS");

    const tablaMetaInfo = {
      metaTotal: hojaDash.getRange("I24").getDisplayValue() || "$0",
      alcanzado: hojaDash.getRange("H26").getDisplayValue() || "$0",
      porAlcanzar: hojaDash.getRange("I26").getDisplayValue() || "$0",
      mensajeExtra: hojaDash.getRange("H27").getDisplayValue() || "" 
    };

    const datosG1 = hojaMotor.getRange("A3:B50").getValues();
    const g1Labels = []; const g1Valores = [];
    datosG1.forEach(fila => { if (fila[0] !== "" && fila[1] > 0) { g1Labels.push(fila[0].toString()); g1Valores.push(fila[1]); } });

    const datosG2 = hojaMotor.getRange("C3:D4").getValues();
    const datosG3 = hojaMotor.getRange("E3:E4").getValues();

    return {
      metricas: {
        totalCobrado: hojaDash.getRange("B4").getValue() || 0,
        facturasPendientes: hojaDash.getRange("E4").getValue() || 0,
        proyectosActivos: hojaDash.getRange("H4").getValue() || 0,
        mes: hojaDash.getRange("A7").getDisplayValue() || ""
      },
      g1: { labels: g1Labels, valores: g1Valores },
      g2: { valores: [(parseFloat(datosG2[0][0])||0)+(parseFloat(datosG2[0][1])||0), (parseFloat(datosG2[1][0])||0)+(parseFloat(datosG2[1][1])||0)] },
      g3: { valores: [parseFloat(datosG3[0][0])||0, parseFloat(datosG3[1][0])||0], infoExtra: tablaMetaInfo },
      proyectos: hojaDash.getRange("B30:H34").getValues().filter(f => f[0] !== "" && f[0] !== "No hay pendientes").map(f => [f[0], f[3], f[6]])
    };
  } catch (e) { return { error: e.toString() }; }
}

    const uF = hojaP.getLastRow();
    if (uF < 3) return { proyectos: [], clientes: [] };
    const datos = hojaP.getRange(3, 1, uF - 2, 13).getValues();
    const clientes = hojaC.getRange(2, 1, Math.max(hojaC.getLastRow()-1, 1), 1).getValues().flat();
    
    const datosFormateados = datos.map(f => f.map(c => (c instanceof Date) ? Utilities.formatDate(c, Session.getScriptTimeZone(), "yyyy-MM-dd") : c));
    return { proyectos: datosFormateados, clientes: clientes.filter(c => c !== "") };
  } catch (e) { return { error: e.toString() }; }
}

function guardarCambioProyecto(index, tipo, valor) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hoja = ss.getSheetByName("PROYECTOS Y FINANZAS");
    const fila = parseInt(index) + 3; 
    
    const columnas = { 'proyecto': 1, 'cliente': 2, 'monto': 3, 'cobrado': 4, 'f_rec': 5, 'f_entr': 6, 'f_pago': 7, 'en_proceso': 8, 'terminado': 9, 'pausado': 10 };
    
    if (columnas[tipo]) {
      if (valor === "" && ['f_rec', 'f_entr', 'f_pago'].includes(tipo)) {
         hoja.getRange(fila, columnas[tipo]).clearContent();
      } else {
         hoja.getRange(fila, columnas[tipo]).setValue(valor);
      }
      SpreadsheetApp.flush(); 
      
      if (tipo === 'f_pago' || tipo === 'cobrado') {
        actualizarEstadosLogicosVencimiento();
        actualizarCalculosDashboard();
      }
      return "✅";
    }
    return "❓";
  } catch (e) { return "❌ Error: " + e.toString(); }
}

function actualizarEstadosLogicosVencimiento() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const hojaP = ss.getSheetByName("PROYECTOS Y FINANZAS");
  if (!hojaP) return;

  const uF = hojaP.getLastRow();
  if (uF < 3) return;

  const rangoP = hojaP.getRange(3, 1, uF - 2, 13);
  const datosP = rangoP.getValues();
  
  const hoy = new Date();
  hoy.setHours(0,0,0,0);

  let nuevosEstados = [];

  for (let i = 0; i < datosP.length; i++) {
    let fila = datosP[i];
    let cobrado = fila[3]; 
    let fechaPago = fila[6]; 
    
    let estadoK = fila[10]; 
    let estadoL = fila[11]; 
    let estadoM = fila[12];

    if (fechaPago instanceof Date && cobrado !== true) {
      let fechaComp = new Date(fechaPago);
      fechaComp.setHours(0,0,0,0);
      let difDias = Math.floor((hoy - fechaComp) / (1000 * 60 * 60 * 24));

      if (difDias >= 0 && (estadoK === "" || estadoK === null)) estadoK = "VENCIDO";
      if (difDias >= 7 && (estadoL === "" || estadoL === null)) estadoL = "VENCIDO 7D";
      if (difDias >= 15 && (estadoM === "" || estadoM === null)) estadoM = "VENCIDO 15D";
    }

    if (cobrado === true) {
      if (estadoK.toString().includes("VENCIDO")) estadoK = "";
      if (estadoL.toString().includes("VENCIDO")) estadoL = "";
      if (estadoM.toString().includes("VENCIDO")) estadoM = "";
    }
    nuevosEstados.push([estadoK, estadoL, estadoM]);
  }
  hojaP.getRange(3, 11, nuevosEstados.length, 3).setValues(nuevosEstados);
}

function ejecutarSistemaDeCobro() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const hojaP = ss.getSheetByName("PROYECTOS Y FINANZAS");
  const hojaC = ss.getSheetByName("CLIENTES");
  const hojaScripts = ss.getSheetByName("SCRIPTS DE COBRO");
  
  if (!hojaP || !hojaC || !hojaScripts) return 0;

  SpreadsheetApp.flush();
  
  const dClientes = hojaC.getRange(2, 1, Math.max(hojaC.getLastRow() - 1, 1), 2).getValues();
  let mapaCorreos = {};
  dClientes.forEach(f => {
    if (f[0]) mapaCorreos[f[0].toString().trim()] = f[1];
  });

  const datosP = hojaP.getDataRange().getValues();
  const scripts = [
    hojaScripts.getRange("B6").getValue(), 
    hojaScripts.getRange("B14").getValue(), 
    hojaScripts.getRange("B22").getValue()
  ];
  
  let conteo = 0;

  for (let i = 2; i < datosP.length; i++) {
    let fila = datosP[i]; 
    let msgIndex = -1; 
    let colARecordar = -1;

    if (fila[10] === "ENVIAR" || fila[10] === "VENCIDO") { msgIndex = 0; colARecordar = 11; }
    else if (fila[11] === "ENVIAR 7D" || fila[11] === "VENCIDO 7D") { msgIndex = 1; colARecordar = 12; }
    else if (fila[12] === "ENVIAR 15D" || fila[12] === "VENCIDO 15D") { msgIndex = 2; colARecordar = 13; }

    if (msgIndex !== -1) {
      let clienteNombre = fila[1] ? fila[1].toString().trim() : "";
      let email = mapaCorreos[clienteNombre];
      
      if (email && email.includes("@")) {
        let cuerpo = scripts[msgIndex]
          .replace("[Nombre]", fila[1])
          .replace("[Proyecto]", fila[0])
          .replace("[Monto]", fila[2]);
        try {
          MailApp.sendEmail(email, "⚠️ Recordatorio de Pago: " + (fila[0] || "Pendiente"), cuerpo);
          hojaP.getRange(i + 1, colARecordar).setValue("ENVIADO"); 
          conteo++;
        } catch (e) { Logger.log("Error envío: " + e.message); }
      }
    }
  }
  return conteo;
}

function botonSheetEjecutarCobro() {
  const ui = SpreadsheetApp.getUi();
  const respuesta = ui.alert('🚀 SISTEMA DE COBRO', '¿Deseas procesar vencimientos y enviar correos?', ui.ButtonSet.YES_NO);
  if (respuesta == ui.Button.YES) {
    actualizarEstadosLogicosVencimiento();
    const total = ejecutarSistemaDeCobro();
    ui.alert('RESULTADO', total > 0 ? '¡Éxito! Se enviaron ' + total + ' correos.' : 'No hay correos pendientes.', ui.ButtonSet.OK);
  }
}

/**
 * 8. DISPARADORES (Triggers UNIFICADOS)
 */
function onEdit(e) {
  if (!e) return;
  const ss = e.source;
  const hojaActiva = ss.getActiveSheet();
  const nombreHoja = hojaActiva.getName();
  const fila = e.range.getRow();
  const col = e.range.getColumn();

  if (fila < 2) return;

  if (nombreHoja === "CLIENTES") {
    hojaActiva.getRange(2, 1, hojaActiva.getLastRow() - 1, 8).sort({column: 1, ascending: true});
  }

  if (nombreHoja === "PROYECTOS Y FINANZAS") {
    if (col === 4 || col === 7) {
      actualizarEstadosLogicosVencimiento();
    }
    if (col === 4 && e.range.getValue() === true) {
      hojaActiva.getRange(fila, 11, 1, 3).clearContent();
    }
    actualizarCalculosDashboard();
  }
  
  if (nombreHoja === "MI PERFIL") {
    actualizarCalculosDashboard();
  }
}

/**
 * 9. REPORTE MENSUAL Y ANUAL (ORIGINAL INTACTO)
 */
function obtenerDatosReporteAnual(anio) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hojaP = ss.getSheetByName("PROYECTOS Y FINANZAS");
    if (!hojaP) return { error: "No existe hoja de proyectos" };

    const datos = hojaP.getRange(3, 1, hojaP.getLastRow() - 2, 7).getValues();
    let mesesTotales = new Array(12).fill(0);
    let totalAnio = 0;

    datos.forEach(fila => {
      let monto = parseFloat(fila[2]) || 0;
      let cobrado = (fila[3] === true || fila[3].toString().toUpperCase() === "TRUE");
      let fecha = fila[6]; // Fecha de pago

      if (cobrado && fecha instanceof Date && !isNaN(fecha)) {
        if (fecha.getFullYear() == anio) {
          let mes = fecha.getMonth(); // 0 = Enero, 11 = Diciembre
          mesesTotales[mes] += monto;
          totalAnio += monto;
        }
      }
    });

    return { meses: mesesTotales, total: totalAnio };
  } catch(e) { return { error: e.toString() }; }
}

function obtenerDatosReporteClientes(anio, mesTexto) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hojaP = ss.getSheetByName("PROYECTOS Y FINANZAS");
    if (!hojaP) return { error: "No existe hoja" };

    const mesesArray = ["ENERO", "FEBRERO", "MARZO", "ABRIL", "MAYO", "JUNIO", "JULIO", "AGOSTO", "SEPTIEMBRE", "OCTUBRE", "NOVIEMBRE", "DICIEMBRE"];
    let mesIndex = mesesArray.indexOf(mesTexto.toUpperCase());
    
    const datos = hojaP.getRange(3, 1, hojaP.getLastRow() - 2, 7).getValues();
    let clientesMap = {};

    datos.forEach(fila => {
      let cliente = fila[1] || "Sin Cliente";
      let monto = parseFloat(fila[2]) || 0;
      let cobrado = (fila[3] === true || fila[3].toString().toUpperCase() === "TRUE");
      let fecha = fila[6];

      if (cobrado && fecha instanceof Date && !isNaN(fecha)) {
        if (fecha.getFullYear() == anio && fecha.getMonth() == mesIndex) {
          if(!clientesMap[cliente]) clientesMap[cliente] = 0;
          clientesMap[cliente] += monto;
        }
      }
    });

    return { labels: Object.keys(clientesMap), valores: Object.values(clientesMap) };
  } catch(e) { return { error: e.toString() }; }
}
