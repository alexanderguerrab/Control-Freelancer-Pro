/**
 * ==========================================
 * RUTADOR API HTTP (PARA CONEXIÓN CON VERCEL)
 * ==========================================
 */
function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const accion = data.accion;
    let respuesta = {};

    switch (accion) {
      case 'verificarUsuario': respuesta = verificarUsuario(data.uid, data.correo); break;
      case 'registrarUsuario': respuesta = registrarUsuario(data.uid, data.correo, data.telefono); break;
      case 'obtenerUsuariosAdmin': respuesta = obtenerUsuariosAdmin(); break;
      case 'actualizarEstatusUsuario': respuesta = actualizarEstatusUsuario(data.uid, data.rol, data.estatus); break;
      case 'eliminarUsuarioServidor': respuesta = eliminarUsuarioServidor(data.uid); break;
      case 'registrarPagoFinanzas': respuesta = registrarPagoFinanzas(data.pago); break;
      case 'obtenerDatosFinanzas': respuesta = obtenerDatosFinanzas(); break;
      case 'obtenerMiSuscripcion': respuesta = obtenerMiSuscripcion(data.correo); break;
      case 'obtenerDatosPerfil': respuesta = obtenerDatosPerfil(); break;
      case 'actualizarDatosPerfilCompleto': respuesta = actualizarDatosPerfilCompleto(data.perfil); break;
      case 'obtenerListaClientes': respuesta = obtenerListaClientes(); break;
      case 'guardarNuevoCliente': respuesta = guardarNuevoCliente(data.cliente); break;
      case 'actualizarClienteServidor': respuesta = actualizarClienteServidor(data.cliente); break;
      case 'eliminarClienteServidor': respuesta = eliminarClienteServidor(data.dni); break;
      case 'obtenerDatosDashboard': respuesta = obtenerDatosDashboard(); break;
      case 'obtenerDatosProyectos': respuesta = obtenerDatosProyectos(); break;
      case 'guardarCambioProyecto': respuesta = guardarCambioProyecto(data.index, data.tipo, data.valor); break;
      case 'obtenerScripts': respuesta = obtenerScripts(); break;
      case 'actualizarScripts': respuesta = actualizarScripts(data.scripts); break;
      case 'ejecutarEnvioDesdeApp': respuesta = { mensaje: ejecutarEnvioDesdeApp() }; break;
      case 'obtenerDatosReporteAnual': respuesta = obtenerDatosReporteAnual(data.anio); break;
      case 'obtenerDatosReporteClientes': respuesta = obtenerDatosReporteClientes(data.anio, data.mesTexto); break;
      default: respuesta = { error: "Acción no reconocida" };
    }

    return ContentService.createTextOutput(JSON.stringify(respuesta))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

/**
 * ==========================================
 * SISTEMA DE CLONACIÓN DE PLANTILLA
 * ==========================================
 */
function crearCopiaPlantillaParaUsuario(correoUsuario) {
  try {
    const archivoPlantilla = DriveApp.getFileById(ID_PLANTILLA_MAESTRA);
    const carpetaDestino = DriveApp.getFolderById(ID_CARPETA_DESTINO);
    
    const nombreCopia = "Dashboard Freelancer - " + correoUsuario;
    const nuevaCopia = archivoPlantilla.makeCopy(nombreCopia, carpetaDestino);
    
    // Asignar permisos de edición al nuevo usuario
    try {
      nuevaCopia.addEditor(correoUsuario);
    } catch(eErr) {
      Logger.log("No se pudo agregar permisos de editor: " + eErr.toString());
    }

    return {
      id: nuevaCopia.getId(),
      url: nuevaCopia.getUrl()
    };
  } catch (e) {
    Logger.log("Error al crear copia de plantilla: " + e.toString());
    return { id: "", url: "" };
  }
}

/**
 * ==========================================
 * SISTEMA DE USUARIOS Y AUTENTICACIÓN (FIREBASE)
 * ==========================================
 */
function verificarUsuario(uid, correo) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hoja = ss.getSheetByName("USUARIOS");
    if (!hoja) return { error: "No existe la hoja USUARIOS" };
    
    const datos = hoja.getDataRange().getValues();
    for (let i = 1; i < datos.length; i++) {
      if (datos[i][0] === uid) {
        return { 
          encontrado: true,
          rol: datos[i][3],     // Columna D (Rol)
          estatus: datos[i][4], // Columna E (Estatus)
          hojaUrl: datos[i][6] || "" // Columna G (URL de la plantilla)
        };
      }
    }
    
    // Si entra por Google y no está registrado, se registra y se crea su plantilla
    return registrarUsuario(uid, correo, "Sin teléfono");
  } catch (e) {
    return { error: e.toString() };
  }
}

function registrarUsuario(uid, correo, telefono) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hoja = ss.getSheetByName("USUARIOS");
    const datos = hoja.getDataRange().getValues();
    
    for (let i = 1; i < datos.length; i++) {
      if (datos[i][0] === uid) return { error: "El usuario ya existe." };
    }
    
    // 1. Crear copia de plantilla
    const copiaDrive = crearCopiaPlantillaParaUsuario(correo);

    // 2. Registrar usuario en la hoja
    const fecha = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd HH:mm:ss");
    hoja.appendRow([uid, correo, telefono, "Cliente", "Pendiente", fecha, copiaDrive.url, copiaDrive.id]);
    
    return { 
      encontrado: true, 
      rol: "Cliente", 
      estatus: "Pendiente",
      mensaje: "✅ Cuenta creada y plantilla clonada. Espere autorización." 
    };
  } catch (e) { return { error: "❌ Error al registrar: " + e.toString() }; }
}

/**
 * ==========================================
 * FINANZAS Y SUSCRIPCIONES
 * ==========================================
 */
function registrarPagoFinanzas(pago) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hojaPagos = ss.getSheetByName("HISTORIAL_PAGOS");
    const hojaSuscripciones = ss.getSheetByName("ESTADO_SUSCRIPCIONES");
    
    const id = "PAG-" + new Date().getTime();
    hojaPagos.appendRow([id, pago.cliente, pago.fechaPago, pago.periodo, pago.plan, pago.metodo, pago.monto, pago.recibo]);
    
    // Actualizar o insertar estado de suscripción
    const datosS = hojaSuscripciones.getDataRange().getValues();
    let encontrado = false;
    let proximoPago = new Date(pago.fechaBase);
    
    if (pago.plan.toUpperCase() === "MENSUAL") {
      proximoPago.setMonth(proximoPago.getMonth() + 1);
    } else {
      proximoPago.setFullYear(proximoPago.getFullYear() + 1);
    }
    const fechaProxStr = Utilities.formatDate(proximoPago, Session.getScriptTimeZone(), "yyyy-MM-dd");

    for (let i = 1; i < datosS.length; i++) {
      if (datosS[i][0] === pago.cliente) {
        hojaSuscripciones.getRange(i + 1, 2, 1, 3).setValues([[pago.plan, fechaProxStr, "AL DÍA"]]);
        encontrado = true;
        break;
      }
    }
    if (!encontrado) {
      hojaSuscripciones.appendRow([pago.cliente, pago.plan, fechaProxStr, "AL DÍA"]);
    }
    return "✅ Pago registrado correctamente.";
  } catch(e) { return "❌ Error: " + e.toString(); }
}

function obtenerDatosFinanzas() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hojaS = ss.getSheetByName("ESTADO_SUSCRIPCIONES");
    const hojaP = ss.getSheetByName("HISTORIAL_PAGOS");
    const hojaC = ss.getSheetByName("CLIENTES");
    
    const clientes = hojaC.getRange(2, 1, Math.max(hojaC.getLastRow()-1, 1), 1).getValues().flat().filter(c => c !== "");
    
    let suscripciones = [];
    if(hojaS.getLastRow() > 1) {
      const valsS = hojaS.getDataRange().getValues();
      valsS.shift();
      suscripciones = valsS.map(f => [f[0], f[1], (f[2] instanceof Date ? Utilities.formatDate(f[2], Session.getScriptTimeZone(), "dd/MM/yyyy") : f[2]), f[3]]);
    }

    let historial = [];
    if(hojaP.getLastRow() > 1) {
      const valsP = hojaP.getDataRange().getValues();
      valsP.shift();
      historial = valsP.map(f => [f[1], (f[2] instanceof Date ? Utilities.formatDate(f[2], Session.getScriptTimeZone(), "dd/MM/yyyy") : f[2]), f[3], f[4], f[5], f[6], f[7]]);
    }
    return { clientes: clientes, suscripciones: suscripciones, historial: historial };
  } catch(e) { return { error: e.toString() }; }
}

function obtenerMiSuscripcion(clienteNombre) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hojaS = ss.getSheetByName("ESTADO_SUSCRIPCIONES");
    const hojaP = ss.getSheetByName("HISTORIAL_PAGOS");
    
    let estadoPlan = "Sin plan activo";
    let proximoPago = "---";
    let estadoActual = "PENDIENTE";

    if(hojaS.getLastRow() > 1) {
      const datosS = hojaS.getDataRange().getValues();
      for(let i=1; i<datosS.length; i++) {
        if(datosS[i][0].toString().trim().toLowerCase() === clienteNombre.toLowerCase()) {
          estadoPlan = datosS[i][1];
          proximoPago = datosS[i][2] instanceof Date ? Utilities.formatDate(datosS[i][2], Session.getScriptTimeZone(), "dd/MM/yyyy") : datosS[i][2];
          estadoActual = datosS[i][3];
          break;
        }
      }
    }

    let historialCli = [];
    if(hojaP.getLastRow() > 1) {
      const datosP = hojaP.getDataRange().getValues();
      for(let i=1; i<datosP.length; i++) {
        if(datosP[i][1].toString().trim().toLowerCase() === clienteNombre.toLowerCase()) {
          let fPago = datosP[i][2] instanceof Date ? Utilities.formatDate(datosP[i][2], Session.getScriptTimeZone(), "dd/MM/yyyy") : datosP[i][2];
          historialCli.push([fPago, datosP[i][3], datosP[i][4], datosP[i][5], datosP[i][6]]);
        }
      }
    }
    return { plan: estadoPlan, proximoPago: proximoPago, estado: estadoActual, historial: historialCli };
  } catch(e) { return { error: e.toString() }; }
}

/**
 * 2. GESTIÓN DE PERFIL
 */
function obtenerDatosPerfil() {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hoja = ss.getSheetByName("MI PERFIL");
    if (!hoja) return { error: "No existe la hoja MI PERFIL" };
    
    const rawLogoUrl = hoja.getRange("C11").getValue();
    let finalLogoUrl = rawLogoUrl;

    if (rawLogoUrl && (rawLogoUrl.includes("drive.google.com") || rawLogoUrl.includes("google.com/open"))) {
      try {
        let id = "";
        if (rawLogoUrl.includes("id=")) {
          id = rawLogoUrl.split("id=")[1].split("&")[0];
        } else if (rawLogoUrl.includes("/d/")) {
          id = rawLogoUrl.split("/d/")[1].split("/")[0];
        } else if (rawLogoUrl.includes("/file/d/")) {
          id = rawLogoUrl.split("/file/d/")[1].split("/")[0];
        }
        if (id) finalLogoUrl = "https://drive.google.com/thumbnail?sz=w500&id=" + id;
      } catch (err) { console.error("Error logo: " + err); }
    }
    
    return {
      nombre: hoja.getRange("C5").getValue(),
      marca: hoja.getRange("C7").getValue(),
      especialidad: hoja.getRange("C9").getValue(),
      logoUrl: finalLogoUrl, 
      moneda: hoja.getRange("C17").getValue(),
      meta: hoja.getRange("C19").getValue(),
      rs1_n: hoja.getRange("B23").getValue(), rs1_l: hoja.getRange("C23").getValue(),
      rs2_n: hoja.getRange("B25").getValue(), rs2_l: hoja.getRange("C25").getValue(),
      rs3_n: hoja.getRange("B27").getValue(), rs3_l: hoja.getRange("C27").getValue(),
      p1_n: hoja.getRange("B31").getValue(), p1_l: hoja.getRange("C31").getValue(),
      p2_n: hoja.getRange("B33").getValue(), p2_l: hoja.getRange("C33").getValue(),
      p3_n: hoja.getRange("B35").getValue(), p3_l: hoja.getRange("C35").getValue()
    };
  } catch (e) { return { error: e.toString() }; }
}

/**
 * 4. DASHBOARD Y MÉTRICAS
 */
function actualizarCalculosDashboard() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const hojaDash = ss.getSheetByName("DASHBOARD");
  const hojaP = ss.getSheetByName("PROYECTOS Y FINANZAS");
  const hojaPerfil = ss.getSheetByName("MI PERFIL");
  const hojaMotor = ss.getSheetByName("MOTOR GRAFICOS");
  
  if (!hojaDash || !hojaP || !hojaPerfil || !hojaMotor) return;

  const hoy = new Date();
  const mesActual = hoy.getMonth();
  const anioActual = hoy.getFullYear();

  const nombrePerfil = hojaPerfil.getRange("C5").getValue().toString().toUpperCase();
  hojaDash.getRange("A1").setValue("DASHBOARD DE " + nombrePerfil);
  
  const nombreMes = hoy.toLocaleDateString('es-ES', { month: 'long' }).toUpperCase();
  hojaDash.getRange("A7").setValue(nombreMes + "-" + anioActual);

  const uF = hojaP.getLastRow();
  let totalCobradoMes = 0;
  let facturasPendMes = 0;
  let proyectosActivosMes = 0;
  let pendientesLista = []; 

  if (uF >= 3) {
    const datosP = hojaP.getRange(3, 1, uF - 2, 7).getValues(); 
    datosP.forEach(fila => {
      const nombreProyecto = fila[0];
      const cliente = fila[1];
      const monto = parseFloat(fila[2]) || 0;
      const cobrado = (fila[3] === true || fila[3].toString().toUpperCase() === "TRUE"); 
      const fecha = fila[6];

      if (fecha instanceof Date && !isNaN(fecha)) {
        if (fecha.getMonth() === mesActual && fecha.getFullYear() === anioActual) {
          proyectosActivosMes++;
          if (cobrado === true) totalCobradoMes += monto;
          else facturasPendMes += monto;
        }
      }

      if (nombreProyecto !== "" && cobrado === false && pendientesLista.length < 5) {
        pendientesLista.push([nombreProyecto, cliente, monto]);
      }
    });
  }

  hojaDash.getRange("B4").setValue(totalCobradoMes);
  hojaDash.getRange("E4").setValue(facturasPendMes);
  hojaDash.getRange("H4").setValue(proyectosActivosMes);

  hojaDash.getRange("B30:B34").clearContent();
  hojaDash.getRange("E30:E34").clearContent();
  hojaDash.getRange("H30:H34").clearContent();

  if (pendientesLista.length > 0) {
    hojaDash.getRange(30, 2, pendientesLista.length, 1).setValues(pendientesLista.map(p => [p[0]]));
    hojaDash.getRange(30, 5, pendientesLista.length, 1).setValues(pendientesLista.map(p => [p[1]]));
    hojaDash.getRange(30, 8, pendientesLista.length, 1).setValues(pendientesLista.map(p => [p[2]]));
  } else {
    hojaDash.getRange("B30").setValue("No hay pendientes");
  }

  const metaConfigurada = parseFloat(hojaPerfil.getRange("C19").getValue()) || 0;
  hojaDash.getRange("I24").setValue(metaConfigurada);
  hojaDash.getRange("H26").setValue(totalCobradoMes);

  if (totalCobradoMes > metaConfigurada) {
    hojaDash.getRange("I26").setValue("¡META SUPERADA!");
  } else {
    hojaDash.getRange("I26").setValue(metaConfigurada - totalCobradoMes);
  }

  const gananciaExtra = Math.max(0, totalCobradoMes - metaConfigurada);
  if (gananciaExtra > 0) {
    hojaDash.getRange("H27").setValue("¡FELICIDADES! HAS GANADO $" + gananciaExtra.toFixed(2) + " EXTRAS ESTE MES 🚀");
  } else {
    hojaDash.getRange("H27").setValue("");
  }

  hojaMotor.getRange("E3").setValue(Math.min(totalCobradoMes, metaConfigurada));
  hojaMotor.getRange("E4").setValue(Math.max(0, metaConfigurada - totalCobradoMes));

  SpreadsheetApp.flush();
}

function guardarCambioProyecto(index, tipo, valor) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hoja = ss.getSheetByName("PROYECTOS Y FINANZAS");
    const fila = parseInt(index) + 3; 
    
    const columnas = { 'proyecto': 1, 'cliente': 2, 'monto': 3, 'cobrado': 4, 'f_rec': 5, 'f_entr': 6, 'f_pago': 7, 'en_proceso': 8, 'terminado': 9, 'pausado': 10 };
    
    if (columnas[tipo]) {
      if (valor === "" && ['f_rec', 'f_entr', 'f_pago'].includes(tipo)) {
         hoja.getRange(fila, columnas[tipo]).clearContent();
      } else {
         hoja.getRange(fila, columnas[tipo]).setValue(valor);
      }
      SpreadsheetApp.flush(); 
      
      if (tipo === 'f_pago' || tipo === 'cobrado') {
        actualizarEstadosLogicosVencimiento();
        actualizarCalculosDashboard();
      }
      return "✅";
    }
    return "❓";
  } catch (e) { return "❌ Error: " + e.toString(); }
}

function actualizarEstadosLogicosVencimiento() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const hojaP = ss.getSheetByName("PROYECTOS Y FINANZAS");
  if (!hojaP) return;

  const uF = hojaP.getLastRow();
  if (uF < 3) return;

  const rangoP = hojaP.getRange(3, 1, uF - 2, 13);
  const datosP = rangoP.getValues();
  
  const hoy = new Date();
  hoy.setHours(0,0,0,0);

  let nuevosEstados = [];

  for (let i = 0; i < datosP.length; i++) {
    let fila = datosP[i];
    let cobrado = fila[3]; 
    let fechaPago = fila[6]; 
    
    let estadoK = fila[10]; 
    let estadoL = fila[11]; 
    let estadoM = fila[12];

    if (fechaPago instanceof Date && cobrado !== true) {
      let fechaComp = new Date(fechaPago);
      fechaComp.setHours(0,0,0,0);
      let difDias = Math.floor((hoy - fechaComp) / (1000 * 60 * 60 * 24));

      if (difDias >= 0 && (estadoK === "" || estadoK === null)) estadoK = "VENCIDO";
      if (difDias >= 7 && (estadoL === "" || estadoL === null)) estadoL = "VENCIDO 7D";
      if (difDias >= 15 && (estadoM === "" || estadoM === null)) estadoM = "VENCIDO 15D";
    }

    if (cobrado === true) {
      if (estadoK.toString().includes("VENCIDO")) estadoK = "";
      if (estadoL.toString().includes("VENCIDO")) estadoL = "";
      if (estadoM.toString().includes("VENCIDO")) estadoM = "";
    }
    nuevosEstados.push([estadoK, estadoL, estadoM]);
  }
  hojaP.getRange(3, 11, nuevosEstados.length, 3).setValues(nuevosEstados);
}

function ejecutarSistemaDeCobro() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const hojaP = ss.getSheetByName("PROYECTOS Y FINANZAS");
  const hojaC = ss.getSheetByName("CLIENTES");
  const hojaScripts = ss.getSheetByName("SCRIPTS DE COBRO");
  
  if (!hojaP || !hojaC || !hojaScripts) return 0;

  SpreadsheetApp.flush();
  
  const dClientes = hojaC.getRange(2, 1, Math.max(hojaC.getLastRow() - 1, 1), 2).getValues();
  let mapaCorreos = {};
  dClientes.forEach(f => {
    if (f[0]) mapaCorreos[f[0].toString().trim()] = f[1];
  });

  const datosP = hojaP.getDataRange().getValues();
  const scripts = [
    hojaScripts.getRange("B6").getValue(), 
    hojaScripts.getRange("B14").getValue(), 
    hojaScripts.getRange("B22").getValue()
  ];
  
  let conteo = 0;

  for (let i = 2; i < datosP.length; i++) {
    let fila = datosP[i]; 
    let msgIndex = -1; 
    let colARecordar = -1;

    if (fila[10] === "ENVIAR" || fila[10] === "VENCIDO") { msgIndex = 0; colARecordar = 11; }
    else if (fila[11] === "ENVIAR 7D" || fila[11] === "VENCIDO 7D") { msgIndex = 1; colARecordar = 12; }
    else if (fila[12] === "ENVIAR 15D" || fila[12] === "VENCIDO 15D") { msgIndex = 2; colARecordar = 13; }

    if (msgIndex !== -1) {
      let clienteNombre = fila[1] ? fila[1].toString().trim() : "";
      let email = mapaCorreos[clienteNombre];
      
      if (email && email.includes("@")) {
        let cuerpo = scripts[msgIndex]
          .replace("[Nombre]", fila[1])
          .replace("[Proyecto]", fila[0])
          .replace("[Monto]", fila[2]);
        try {
          MailApp.sendEmail(email, "⚠️ Recordatorio de Pago: " + (fila[0] || "Pendiente"), cuerpo);
          hojaP.getRange(i + 1, colARecordar).setValue("ENVIADO"); 
          conteo++;
        } catch (e) { Logger.log("Error envío: " + e.message); }
      }
    }
  }
  return conteo;
}

function botonSheetEjecutarCobro() {
  const ui = SpreadsheetApp.getUi();
  const respuesta = ui.alert('🚀 SISTEMA DE COBRO', '¿Deseas procesar vencimientos y enviar correos?', ui.ButtonSet.YES_NO);
  if (respuesta == ui.Button.YES) {
    actualizarEstadosLogicosVencimiento();
    const total = ejecutarSistemaDeCobro();
    ui.alert('RESULTADO', total > 0 ? '¡Éxito! Se enviaron ' + total + ' correos.' : 'No hay correos pendientes.', ui.ButtonSet.OK);
  }
}

/**
 * 8. DISPARADORES (Triggers UNIFICADOS)
 */
function onEdit(e) {
  if (!e) return;
  const ss = e.source;
  const hojaActiva = ss.getActiveSheet();
  const nombreHoja = hojaActiva.getName();
  const fila = e.range.getRow();
  const col = e.range.getColumn();

  if (fila < 2) return;

  if (nombreHoja === "CLIENTES") {
    hojaActiva.getRange(2, 1, hojaActiva.getLastRow() - 1, 8).sort({column: 1, ascending: true});
  }

  if (nombreHoja === "PROYECTOS Y FINANZAS") {
    if (col === 4 || col === 7) {
      actualizarEstadosLogicosVencimiento();
    }
    if (col === 4 && e.range.getValue() === true) {
      hojaActiva.getRange(fila, 11, 1, 3).clearContent();
    }
    actualizarCalculosDashboard();
  }
  
  if (nombreHoja === "MI PERFIL") {
    actualizarCalculosDashboard();
  }
}

/**
 * 9. REPORTE MENSUAL Y ANUAL
 */
function obtenerDatosReporteAnual(anio) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hojaP = ss.getSheetByName("PROYECTOS Y FINANZAS");
    if (!hojaP) return { error: "No existe hoja de proyectos" };

    const datos = hojaP.getRange(3, 1, hojaP.getLastRow() - 2, 7).getValues();
    let mesesTotales = new Array(12).fill(0);
    let totalAnio = 0;

    datos.forEach(fila => {
      let monto = parseFloat(fila[2]) || 0;
      let cobrado = (fila[3] === true || fila[3].toString().toUpperCase() === "TRUE");
      let fecha = fila[6];

      if (cobrado && fecha instanceof Date && !isNaN(fecha)) {
        if (fecha.getFullYear() == anio) {
          let mes = fecha.getMonth();
          mesesTotales[mes] += monto;
          totalAnio += monto;
        }
      }
    });

    return { meses: mesesTotales, total: totalAnio };
  } catch(e) { return { error: e.toString() }; }
}

function obtenerDatosReporteClientes(anio, mesTexto) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const hojaP = ss.getSheetByName("PROYECTOS Y FINANZAS");
    if (!hojaP) return { error: "No existe hoja" };

    const mesesArray = ["ENERO", "FEBRERO", "MARZO", "ABRIL", "MAYO", "JUNIO", "JULIO", "AGOSTO", "SEPTIEMBRE", "OCTUBRE", "NOVIEMBRE", "DICIEMBRE"];
    let mesIndex = mesesArray.indexOf(mesTexto.toUpperCase());
    
    const datos = hojaP.getRange(3, 1, hojaP.getLastRow() - 2, 7).getValues();
    let clientesMap = {};

    datos.forEach(fila => {
      let cliente = fila[1] || "Sin Cliente";
      let monto = parseFloat(fila[2]) || 0;
      let cobrado = (fila[3] === true || fila[3].toString().toUpperCase() === "TRUE");
      let fecha = fila[6];

      if (cobrado && fecha instanceof Date && !isNaN(fecha)) {
        if (fecha.getFullYear() == anio && fecha.getMonth() == mesIndex) {
          if(!clientesMap[cliente]) clientesMap[cliente] = 0;
          clientesMap[cliente] += monto;
        }
      }
    });

    return { labels: Object.keys(clientesMap), valores: Object.values(clientesMap) };
  } catch(e) { return { error: e.toString() }; }
}
